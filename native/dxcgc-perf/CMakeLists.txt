cmake_minimum_required(VERSION 3.24)
project(winml_dxcgc_perf LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "DXCGC perf requires Windows")
endif()
find_path(DXCGC_INCLUDE_DIR dxcgc.h REQUIRED)
add_executable(winml-dxcgc-perf main.cpp)
target_compile_features(winml-dxcgc-perf PRIVATE cxx_std_17)
target_include_directories(winml-dxcgc-perf PRIVATE ${DXCGC_INCLUDE_DIR})
target_link_libraries(winml-dxcgc-perf PRIVATE d3d12 dxgi dxguid)
