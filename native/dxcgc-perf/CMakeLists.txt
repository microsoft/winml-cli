cmake_minimum_required(VERSION 3.24)
project(winml_dxcgc_perf LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "DXCGC perf requires Windows")
endif()
find_path(DXCGC_INCLUDE_DIR dxcgc.h REQUIRED)
add_executable(winml-dxcgc-perf main.cpp)
target_compile_features(winml-dxcgc-perf PRIVATE cxx_std_17)
target_include_directories(winml-dxcgc-perf PRIVATE ${DXCGC_INCLUDE_DIR})
target_link_libraries(winml-dxcgc-perf PRIVATE d3d12 dxgi dxguid)
add_library(winml-dxcgc-python SHARED main.cpp)
target_compile_features(winml-dxcgc-python PRIVATE cxx_std_17)
target_include_directories(winml-dxcgc-python PRIVATE ${DXCGC_INCLUDE_DIR})
target_link_libraries(winml-dxcgc-python PRIVATE d3d12 dxgi dxguid)
include(CTest)
if(BUILD_TESTING)
  add_executable(constant-export-test constant_export_test.cpp)
  target_compile_features(constant-export-test PRIVATE cxx_std_17)
  add_test(NAME constant-export COMMAND constant-export-test)
endif()
